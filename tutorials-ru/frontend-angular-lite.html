Angular Lite: свой фреймворк на JavaScript

1. Три столпа AngularJS

Статья выделяет три главные части и восстанавливает их по одной:

  • DOMCompiler ($compile): находит директивы (directives) в HTML и выполняет их;
  • Provider ($provide + $injector): определяет и внедряет сервисы;
  • Scope: иерархическая модель, делающая возможным двустороннее связывание.

Именно это и делает AngularJS AngularJS'ом; всё остальное (роутинг, фильтры, элементы форм) просто собрано вокруг этих трёх осей.

2. DOMCompiler: поиск и выполнение директив

Когда AngularJS загружается, он начинает «компилировать» DOM: обходит все элементы, находит директивы по имени атрибута и выполняет их с текущим scope. Наша простая реализация:

jsconst DOMCompiler = {
  bootstrap() {
    // рекурсивный обход всех DOM-узлов
    this.compile(document.children[0], Provider.get('$rootScope'));
  },

  compile(el, scope) {
    const dirs = this._getElDirectives(el);
    // выполнение каждой директивы со scope
    dirs.forEach((d) => {
      d.scope = scope;
      d.fn(scope, el, d.attrs, d);
    });
    // атрибут ng-repeat и ему подобные: игнорируем повторяемый элемент
    if (dirs.some((d) => d.name === 'ng-repeat')) return;
    // обход детей
    Array.prototype.slice.call(el.children).forEach((child) => {
      this.compile(child, scope);
    });
  },

  _getElDirectives(el) {
    const attrs = el.attributes;
    const result = [];
    for (let i = 0; i < attrs.length; i++) {
      const attr = attrs[i];
      if (Directive.get(attr.nodeName)) {
        result.push({
          name: attr.nodeName,
          value: attr.nodeValue,
          fn: Directive.get(attr.nodeName),
        });
      }
    }
    return result;
  },
};

Директивы хранятся в реестре (Directive): это функции, принимающие (scope, element, attrs) и обычно выполняющие «одноразовые» операции — например, регистрацию watch или построение нового DOM.

3. Provider и внедрение зависимостей

Внедрение зависимостей в AngularJS просто: сервисы регистрируются по имени, и когда функция объявляет «хочу такой-то сервис», injector находит и доставляет его. Имена аргументов функции и есть «описание потребности»:

jsconst Provider = {
  _providers: {},
  _cache: {},

  provide(name, fn) {
    this._providers[name] = fn;
  },

  // внедрение: читаем имена аргументов функции из её определения
  get(name) {
    if (this._cache[name]) return this._cache[name];
    const provider = this._providers[name];
    if (!provider) throw new Error('Неизвестный сервис: ' + name);

    // получаем имена параметров, превратив функцию в строку и разобрав её текст
    const args = this._getParams(provider);
    // автоматическое создание зависимых сервисов (рекурсивно)
    const deps = args.map((arg) => this.get(arg));
    const instance = provider.apply(null, deps);
    this._cache[name] = instance; // только один экземпляр каждого сервиса
    return instance;
  },

  _getParams(fn) {
    // разбор строки определения функции для извлечения имён параметров
    const fnStr = fn.toString();
    const params = fnStr.match(/^function\s*[^\(]*\(\s*([^\)]*)\)/m);
    if (!params || !params[1]) return [];
    return params[1].split(',').map((p) => p.trim().replace(/[^_$a-zA-Z0-9]/g, ''));
  },
};

Тот же приём, что использовал сам AngularJS (а Angular 2+ — с компилируемым DI): «имена параметров» были проблемой в продакшене (минификация), и поэтому Angular позже перешёл на явное внедрение (комментарий или массив имён).

4. Scope и двустороннее связывание

Каждый scope — дерево: через $new() создаётся дочерний scope (который видит свойства родителя, но не возвращает ему свои изменения), а $digest запускает цикл проверки изменений. Директива ng-bind привязывает значение scope к тексту элемента:

jsclass Scope {
  constructor() {
    this.$$watchers = [];
  }

  $watch(fn, listener) {
    const watcher = { fn, last: undefined, listener };
    this.$$watchers.push(watcher);
  }

  // цикл digest: повторяй проверку, пока есть изменения (максимум 10 раз)
  $digest() {
    let dirty = true;
    let count = 0;
    while (dirty && count < 10) {
      dirty = this._digestOnce();
      count++;
    }
  }

  _digestOnce() {
    let dirty = false;
    this.$$watchers.forEach((watcher) => {
      const newValue = watcher.fn(this);
      if (newValue !== watcher.last) {
        watcher.last = newValue;
        watcher.listener(newValue, watcher.last, this);
        dirty = true;
      }
    });
    return dirty;
  }

  $new() {
    // иерархия scope: для дочерних
    const child = new Scope();
    child.$parent = this;
    return child;
  }
}

// директива ng-bind: привязка выражения scope к тексту элемента
Directive.set('ng-bind', (scope, el, attrs) => {
  scope.$watch(() => {
    // вычисление выражения (например "name") в текущем scope
    return eval('scope.' + attrs['ng-bind']);
  }, (value) => {
    el.innerHTML = value;
  });
});

5. Собираем всё вместе: полное приложение

js// регистрация сервисов
Provider.provide('$scope', () => new Scope());
Provider.provide('$rootScope', () => new Scope());

// директива ng-model для полей ввода (двустороннее связывание)
Directive.set('ng-model', (scope, el) => {
  el.addEventListener('input', () => {
    scope.model = el.value;
    scope.$digest(); // digest выполняется при каждом вводе
  });
});

// bootstrap: начинаем с корня документа
DOMCompiler.bootstrap();

Теперь любой HTML с ng-bind="model" и ng-model на поле ввода ведёт себя как настоящий двусторонне связанный AngularJS — и всё это примерно в 200 строк!

Дополнительно: почему AngularJS был таким медленным? Потому что каждое изменение запускало полный digest всего дерева scope (dirty checking). Новые фреймворки, такие как Angular и Vue, вместо этого отслеживают изменения напрямую — через «зоны» или «наблюдение getter/setter».

6. Итоги

  • Директивы = функции, выполняемые со scope; ng-repeat, ng-bind и прочие — всё это записи в простом реестре.
  • Внедрение зависимостей = чтение имён аргументов из текста функции.
  • Scope = дерево watcher'ов с циклом digest; источник и силы, и слабости AngularJS.

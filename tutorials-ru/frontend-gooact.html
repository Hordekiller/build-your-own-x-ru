Gooact: React на 500 строках

1. Введение: цель и ограничения

Цель статьи — реализовать «минимальный React», способный запускать простые настоящие приложения: React.createElement, класс React.Component с setState и механизм diffing для эффективных обновлений. Автор сознательно отказывается от некоторых возможностей (таких как функциональные компоненты, порталы и продвинутые оптимизации), чтобы код оставался маленьким.

Совет: преимущество этого учебника — краткость; можно прочитать весь код сразу, а затем перейти к более полной версии (например, Didact с Fiber и Hooks).

2. createElement: узлы виртуального дерева

Как и во всех версиях, первая деталь — фабрика узлов. Особенность реализации Gooact: узлы напрямую сопоставляются с именами CSS-классов, а события распознаются по префиксу on:

jsconst TEXT_ELEMENT = "TEXT ELEMENT";

function createElement(type, config, ...args) {
  const props = Object.assign({}, config);
  // дети: текстовые узлы превращаются в TEXT_ELEMENT
  const hasChildren = args.length > 0;
  const rawChildren = hasChildren ? [].concat(...args) : [];
  props.children = rawChildren
    .filter(c => c != null && c !== false)
    .map(c => (c instanceof Object ? c : createTextElement(c)));
  return { type, props };
}

function createTextElement(value) {
  return createElement(TEXT_ELEMENT, { nodeValue: value });
}

3. render: превращение виртуального дерева в DOM

Функция render рекурсивно создаёт DOM-узлы и навешивает атрибуты; события (с префиксом on) подключаются через addEventListener, остальные ключи становятся атрибутами:

jsfunction render(element, parentDom) {
  const dom =
    element.type === TEXT_ELEMENT
      ? document.createTextNode("")
      : document.createElement(element.type);

  const isListener = (name) => name.startsWith("on");
  const isAttribute = (name) => !isListener(name) && name !== "children";

  // подключение событий
  Object.keys(element.props || {})
    .filter(isListener)
    .forEach((name) => {
      const eventType = name.toLowerCase().substring(2);
      dom.addEventListener(eventType, element.props[name]);
    });

  // установка атрибутов
  Object.keys(element.props || {})
    .filter(isAttribute)
    .forEach((name) => {
      dom[name] = element.props[name];
    });

  // рекурсивный рендер детей
  const childElements = element.props.children || [];
  childElements.forEach((childElement) => render(childElement, dom));

  parentDom.appendChild(dom);
  return dom;
}

Обратите внимание, как одно лишь правило именования (префикс on) разделяет события и атрибуты — эта простота повышает читаемость кода.

4. Класс Component и setState

Сердце React — класс Component: каждый экземпляр имеет props, state и __internalInstance, указывающий на собственный DOM-узел. При setState перерисовывается только сам узел и его дети (не вся страница):

jsclass Component {
  constructor(props) {
    this.props = props || {};
    this.state = null;
  }

  setState(nextState) {
    // объединение нового state с прежним
    this.state = Object.assign({}, this.state, nextState);
    // перерисовка только этого компонента
    updateInstance(this.__internalInstance);
  }
}

function updateInstance(internalInstance) {
  const parentDom = internalInstance.dom.parentNode;
  const element = internalInstance.element;
  const newElement = getElement(internalInstance);
  const newInstance = instantiate(newElement);
  // сравнение деревьев и применение diff между старым и новым DOM
  diff(parentDom, internalInstance.dom, newInstance.dom, internalInstance.dom);
  internalInstance.dom = newInstance.dom;
}

function instantiate(element) {
  const { type, props } = element;
  // компоненты: создаём экземпляр и заполняем через метод render
  const isDomElement = typeof type === "string";
  if (isDomElement) {
    const isTextElement = type === TEXT_ELEMENT;
    const dom = isTextElement ? document.createTextNode("") : document.createElement(type);
    updateDomProperties(dom, [], props);
    const childElements = props.children || [];
    const childInstances = childElements.map(instantiate);
    const childDoms = childInstances.map((ci) => ci.dom);
    childDoms.forEach((childDom) => dom.appendChild(childDom));
    return { dom, element, childInstances };
  } else {
    // класс-компонент: создаём экземпляр и рендерим
    const instance = new type(props);
    const childElement = instance.render();
    const childInstance = instantiate(childElement);
    const dom = childInstance.dom;
    // привязываем внутренний экземпляр для последующих обновлений
    Object.assign(instance, { __internalInstance: { dom, element, childInstance } });
    return { dom, element, childInstance, instance };
  }
}

5. diffing: умное обновление узлов

Функция diff охватывает четыре случая: перемещение узла (через insertBefore), удаление, замену новым узлом и «тот же узел» — в этом случае сравниваются и применяются только атрибуты и дети:

jsfunction diff(parentDom, instance, newInstance, referenceDom) {
  const oldDom = instance.dom;
  const newDom = newInstance.dom;

  if (instance.element.type === newInstance.element.type) {
    // тот же тип: обновляем только props
    updateDomProperties(oldDom, instance.element.props, newInstance.element.props);
    const oldChildInstances = instance.childInstances;
    const newChildElements = newInstance.element.props.children || [];
    // diff новых детей по количеству прежних
    const newChildInstances = [];
    const count = Math.max(oldChildInstances.length, newChildElements.length);
    for (let i = 0; i < count; i++) {
      const newChildElement = newChildElements[i];
      const oldChildInstance = oldChildInstances[i];
      if (oldChildInstance && newChildElement) {
        // есть оба: рекурсивный diff
        newChildInstances.push(diff(parentDom, oldChildInstance, instantiate(newChildElement), oldChildInstance.dom));
      } else if (oldChildInstance) {
        // было и исчезло: удаляем
        parentDom.removeChild(oldChildInstance.dom);
      } else {
        // новое: добавляем
        parentDom.appendChild(instantiate(newChildElement).dom);
      }
    }
    instance.childInstances = newChildInstances;
    return instance;
  }
  // тип изменился: полная замена
  parentDom.replaceChild(newDom, oldDom);
  return newInstance;
}

Обратите внимание: поскольку дети сравниваются «по индексу», эта версия не поддерживает ключи (key) — одну из оптимизаций, которые добавляет в сравнение настоящий React.

6. Итоги

  • Ядро React (element ← render ← setState ← diff) реализуемо в 160 строках чистого JavaScript без каких-либо накладных расходов.
  • Разделение «DOM-узла» и «внутреннего экземпляра» — именно то, что позволяет обновлять часть дерева без полного перерендера страницы.
  • Diff работает по «типу узла» и «индексу ребёнка»; ключи (key) — способ улучшить это сравнение в настоящем React.

1. Обзор: восемь шагов

Статья строит полный React за восемь шагов — ту самую архитектуру React 16.8: createElement ← render ← конкурентный рендер (Concurrent Mode) ← Fiber ← фазы Render и Commit ← Reconciliation ← функциональные компоненты ← Hooks. В конце у вас единый код, включающий все шаги; краткое изложение всего кода — в конце статьи.

Примечание: код статьи соответствует React 16.8 — никаких propTypes и классов; всё на функциях и хуках.

2. Шаги 1 и 2: createElement и render

Виртуальное дерево строится через createElement, а render превращает его в DOM. Особенности: всё — даже тексты — является узлами, а render создаёт узлы рекурсивно:

jsfunction createElement(type, props, ...children) {
  return {
    type,
    props: {
      ...props,
      children: children.map((child) =>
        typeof child === 'object' ? child : createTextElement(child)
      ),
    },
  };
}

function createTextElement(text) {
  return { type: 'TEXT_ELEMENT', props: { nodeValue: text, children: [] } };
}

function createDom(fiber) {
  const dom =
    fiber.type === 'TEXT_ELEMENT'
      ? document.createTextNode('')
      : document.createElement(fiber.type);

  updateDom(dom, {}, fiber.props);
  return dom;
}

function render(element, container) {
  const rootFiber = {
    dom: container,
    props: { children: [element] },
    alternate: currentRoot, // для сравнения при следующем рендере
  };
  deletions = [];
  wipRoot = rootFiber;
  nextUnitOfWork = rootFiber;
}

3. Шаги 3 и 4: конкурентный рендер и Fiber

Проблема рекурсивного рендера: всё дерево рендерится разом, без остановок, блокируя поток. Решение: разбить работу на мелкие единицы и выполнять их порциями через requestIdleCallback. Структура данных Fiber и есть такая единица работы:

jslet nextUnitOfWork = null;
let wipRoot = null;      // корень строящегося дерева
let currentRoot = null;  // прежнее закоммиченное дерево
let deletions = [];      // удаляемые узлы

// цикл работы: каждый раз, когда браузер простаивает, — одна единица работы
function workLoop(deadline) {
  let shouldYield = false;
  while (nextUnitOfWork && !shouldYield) {
    nextUnitOfWork = performUnitOfWork(nextUnitOfWork);
    shouldYield = deadline.timeRemaining() < 1;
  }
  // когда вся работа завершена, коммитим всё дерево
  if (!nextUnitOfWork && wipRoot) {
    commitRoot();
  }
  requestIdleCallback(workLoop);
}
requestIdleCallback(workLoop);

// каждый fiber: { type, props, dom, parent, child, sibling, alternate, effectTag }
function performUnitOfWork(fiber) {
  const isFunctionComponent = fiber.type instanceof Function;
  if (isFunctionComponent) {
    updateFunctionComponent(fiber); // функциональные компоненты
  } else {
    updateHostComponent(fiber);     // DOM-узлы
  }
  // выбор следующей единицы работы: ребёнок ← сестра ← родитель
  if (fiber.child) return fiber.child;
  let nextFiber = fiber;
  while (nextFiber) {
    if (nextFiber.sibling) return nextFiber.sibling;
    nextFiber = nextFiber.parent;
  }
}

4. Шаги 5 и 6: фазы Render/Commit и Reconciliation

Строящееся дерево (wip) создаётся в фазе render и применяется к DOM разом — в фазе commit — только когда полностью готово. Сравнение нового дерева с прежним (через alternate) выполняется в reconcileChildren:

jsfunction reconcileChildren(wipFiber, elements) {
  let index = 0;
  let oldFiber = wipFiber.alternate && wipFiber.alternate.child;
  let prevSibling = null;

  while (index < elements.length || oldFiber != null) {
    const element = elements[index];
    let newFiber = null;

    const sameType = oldFiber && element && element.type === oldFiber.type;

    if (sameType) {
      // тот же тип: обновление props
      newFiber = {
        type: oldFiber.type,
        props: element.props,
        dom: oldFiber.dom,
        parent: wipFiber,
        alternate: oldFiber,
        effectTag: 'UPDATE',
      };
    } else if (element) {
      // новый элемент: создание узла
      newFiber = {
        type: element.type,
        props: element.props,
        dom: null,
        parent: wipFiber,
        alternate: null,
        effectTag: 'PLACEMENT',
      };
    } else if (oldFiber) {
      // удаляемый узел
      oldFiber.effectTag = 'DELETION';
      deletions.push(oldFiber);
    }

    // переход к сестре и детям ...
    if (oldFiber) oldFiber = oldFiber.sibling;
    if (index === 0) wipFiber.child = newFiber;
    else prevSibling.sibling = newFiber;
    prevSibling = newFiber;
    index++;
  }
}

// коммит: применение effectTag в DOM — один полный проход
function commitRoot() {
  deletions.forEach(commitWork); // сначала удаления
  commitWork(wipRoot.child);
  currentRoot = wipRoot;
  wipRoot = null;
}

function commitWork(fiber) {
  if (!fiber) return;
  let domParentFiber = fiber.parent;
  while (!domParentFiber.dom) {
    domParentFiber = domParentFiber.parent; // у компонентов нет dom
  }
  const domParent = domParentFiber.dom;

  if (fiber.effectTag === 'PLACEMENT' && fiber.dom != null) {
    domParent.appendChild(fiber.dom);
  } else if (fiber.effectTag === 'UPDATE' && fiber.dom != null) {
    updateDom(fiber.dom, fiber.alternate.props, fiber.props);
  } else if (fiber.effectTag === 'DELETION') {
    commitDeletion(fiber, domParent);
  }

  commitWork(fiber.child);
  commitWork(fiber.sibling);
}

5. Шаги 7 и 8: функциональные компоненты и Hooks

Функциональный компонент — это fiber, у которого type является функцией: вместо построения DOM вызывается функция с props, чтобы получить детей. Хуки работают через глобальные переменные wipFiber и hookIndex — именно это объясняет правило «не вызывай хуки условно»:

jslet wipFiber = null;
let hookIndex = null;

function updateFunctionComponent(fiber) {
  wipFiber = fiber;
  hookIndex = 0;
  wipFiber.hooks = [];
  // вызов функции компонента для получения детей
  const children = [fiber.type(fiber.props)];
  reconcileChildren(fiber, children);
}

function useState(initial) {
  const oldHook =
    wipFiber.alternate &&
    wipFiber.alternate.hooks &&
    wipFiber.alternate.hooks[hookIndex];

  const hook = {
    state: oldHook ? oldHook.state : initial,
    queue: [],
  };

  // выполнение всех действий из очереди
  const actions = oldHook ? oldHook.queue : [];
  actions.forEach((action) => {
    hook.state = typeof action === 'function' ? action(hook.state) : action;
  });

  const setState = (action) => {
    hook.queue.push(action);
    // планирование перерисовки от корня
    wipRoot = {
      dom: currentRoot.dom,
      props: currentRoot.props,
      alternate: currentRoot,
    };
    nextUnitOfWork = wipRoot;
    deletions = [];
  };

  wipFiber.hooks.push(hook);
  hookIndex++;
  return [hook.state, setState];
}

Красивый момент: поскольку useState читает прежнее значение из alternate, при каждом рендере state сохраняется и применяются только изменившиеся значения.

6. Итоги

  • React = дерево fiber + цикл работы + сравнение + commit; никакой магии.
  • Конкурентный рендер возможен благодаря разбиению работы на мелкие единицы и requestIdleCallback.
  • Хуки — просто массив state, хранимый по индексу, — и именно поэтому порядок их вызова критичен.

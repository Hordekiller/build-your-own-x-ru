Flappy Bird на JavaScript

Об уроке

Томас Палф (создатель блога lessmilk.com и серии «одна игра в неделю») написал простую версию Flappy Bird на Phaser 2.4 — всего 65 строк кода — и объясняет её шаг за шагом: гравитация, прыжок по клику, движущиеся трубы и столкновения. Обновлённая версия для Phaser 3 есть на lessmilk.com/phaser-game-tutorial.

Замечание: это лучший первый урок по Phaser: вся структура игры (три сцены: Boot/Play/GameOver) показана в минимально возможном коде.

Структура файлов

textflappy-phaser/
├── index.html        # загрузка Phaser и main.js
├── main.js           # вся игра — 65 строк
└── assets/
    ├── bird.png      # птица
    └── pipe.png      # труба

Код игры (полный)

js// Вся игра в 65 строк!
var game = new Phaser.Game(400, 490, Phaser.AUTO, 'gameDiv');

// Главная сцена: пустая, потому что всё создаётся в состоянии Play
var mainState = {
  preload: function() {
    game.load.image('bird', 'assets/bird.png');
    game.load.image('pipe', 'assets/pipe.png');
  },

  create: function() {
    // гравитация: птица каждое мгновение падает вниз
    game.physics.startSystem(Phaser.Physics.ARCADE);

    this.bird = game.add.sprite(100, 245, 'bird');
    game.physics.arcade.enable(this.bird);
    this.bird.body.gravity.y = 1000;

    // прыжок по клику/нажатию
    var spaceKey = game.input.keyboard.addKey(Phaser.Keyboard.SPACEBAR);
    spaceKey.onDown.add(this.jump, this);

    // группа труб + таймер их создания
    this.pipes = game.add.group();
    this.timer = game.time.events.loop(1500, this.addRowOfPipes, this);

    this.score = 0;
    this.labelScore = game.add.text(20, 20, "0",
      { font: "30px Arial", fill: "#ffffff" });
  },

  update: function() {
    // столкновение птицы с трубами → конец игры
    game.physics.arcade.overlap(this.bird, this.pipes, this.restartGame, null, this);

    if (this.bird.y < 0 || this.bird.y > 490) {
      this.restartGame();
    }
  },

  jump: function() {
    this.bird.body.velocity.y = -350; // толчок вверх
  },

  addRowOfPipes: function() {
    // случайный проход (расстояние от 1 до 5)
    var hole = Math.floor(Math.random() * 5) + 1;

    for (var i = 0; i < 10; i++) {
      if (i != hole && i != hole + 1) {
        var pipe = game.add.sprite(400, i * 49, 'pipe');
        game.physics.arcade.enable(pipe);
        pipe.body.velocity.x = -200; // движение влево
        pipe.checkWorldBounds = true;
        pipe.outOfBoundsKill = true;
        this.pipes.add(pipe);
      }
    }
    this.score += 1;
    this.labelScore.text = this.score;
  },

  restartGame: function() {
    game.state.start('main'); // начать заново
  },
};

game.state.add('main', mainState);
game.state.start('main');

Вот и всё! Полноценный Flappy Bird — достаточно открыть index.html, и он заработает.

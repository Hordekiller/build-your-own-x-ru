Компилятор C на OCaml

О серии

Нора Сандлер в 10 подробных статьях строит на OCaml компилятор для большой подмножества языка C: начинается с int main() { return 1; } и доходит до продвинутых конструкций (struct, логические типы, указатели). Выход — ассемблер x86-64, который линкуется с gcc и запускается.

Важнейшие части

Часть 1: lexer и parser для простейшей программы; часть 2: генерация ассемблера для return; часть 3: локальные переменные и присваивание (через выделение на стеке); часть 4: if/else и циклы; часть 5: функции и аргументы; часть 6: полные унарные/бинарные операторы (например, ~ и !); часть 7: указатели (разыменование) и взятие адреса (&); часть 8: struct и union; часть 9: оставшиеся ключевые слова и поддержка typedef; часть 10: продвинутые типы данных (long, unsigned и т.д.).

Пример кода: правило парсера для сложения

ocaml(* Разбор выражения с приоритетом слева *)
let rec parse_exp tokens =
  let left, tokens = parse_exp1 tokens in
  parse_exp_tail left tokens

(* exp-tail := ("+"|"-"|"*"|"/") exp exp-tail | eps *)
and parse_exp_tail left tokens =
  match tokens with
  | Tok_PLUS :: rest ->
      let right, rest' = parse_exp1 rest in
      parse_exp_tail (Exp_Binop (Add, left, right)) rest'
  | Tok_MINUS :: rest ->
      let right, rest' = parse_exp1 rest in
      parse_exp_tail (Exp_Binop (Sub, left, right)) rest'
  | _ -> left, tokens

Почему эта серия особенная?

Реализация на OCaml означает: безупречный pattern matching для AST и более читаемые типы — так что «код компилятора» сам становится уроком типизации. Также в каждой части есть тесты и руководство по отладке; а в конце ваш компилятор переводит реальные программы на C (с оговоренными ограничениями).

Итоги

Полная, читаемая и близкая к реальности серия. Если вам комфортно с OCaml (или вы готовы выучить его по пути) — этот путь отлично подходит для построения «настоящего» (не игрушечного) компилятора. После неё рекомендуется перейти к стандартам через тестирование C.

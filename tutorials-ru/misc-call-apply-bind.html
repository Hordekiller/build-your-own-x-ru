Реализуй сам методы call(), apply() и bind() в JavaScript

Что делают эти три метода?

Все три меняют this функции: call — с отдельными аргументами, apply — с массивом, а bind возвращает новую функцию, к которой this привязывается навсегда. Глубокое понимание этих методов показывает, как работает this в JavaScript.

Реализация myCall

Основная идея проста: вызвать функцию как метод context. Хотя это можно сделать и через Reflect, простая схема с временным свойством нагляднее:

javascript// myCall: вызов функции с нужным this и отдельными аргументами
Function.prototype.myCall = function (context = globalThis, ...args) {
  // Уникальный ключ против коллизий
  const fnKey = Symbol('fn');

  context[fnKey] = this;        // привязываем функцию к context
  const result = context[fnKey](...args);  // вызываем с аргументами
  delete context[fnKey];        // убираем следы
  return result;
};

// Использование
const person = { name: 'Сара' };
function greet(greeting) {
  return `${greeting}, ${this.name}`;
}
console.log(greet.myCall(person, 'Привет')); // Привет, Сара

Заметки: если context равен null, берём globalThis; для уникального ключа используем Symbol, чтобы избежать коллизий.

myApply и myBind

myApply работает с массивом аргументов (или arguments), а myBind должен вернуть новую функцию, объединяющую начальные аргументы:

javascript// myApply: как call, но с массивом аргументов
Function.prototype.myApply = function (context = globalThis, argsArray = []) {
  const fnKey = Symbol('fn');
  context[fnKey] = this;
  const result = context[fnKey](...argsArray);
  delete context[fnKey];
  return result;
};

// myBind: возвращает функцию, привязанную к this
Function.prototype.myBind = function (context = globalThis, ...initialArgs) {
  const original = this;
  return function (...laterArgs) {
    return original.myCall(context, ...initialArgs, ...laterArgs);
  };
};

// Использование bind
const boundGreet = greet.myBind(person);
console.log(boundGreet('Привет'));  // Привет, Сара

myBind создаёт замыкание (closure): original, context и initialArgs сохраняются и комбинируются при каждом вызове.

Зачем их реализовывать?

Пересоздание этих методов превращает механику this и замыканий из «магии» в «концепцию». Обсуждаются и альтернативные подходы — Reflect.apply и WeakMap для context; в конце вы поймёте, как bind устроен внутри движка V8.

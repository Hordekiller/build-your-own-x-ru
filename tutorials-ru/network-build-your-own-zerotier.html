Что такое ZeroTier и зачем собирать свой

ZeroTier — open-source проект, который создаёт виртуальный сетевой интерфейс (Virtual Network Interface) с P2P-соединениями, как локальная сеть; в результате ваши машины могут ping, SSH и REACH друг друга по IP — из любой точки мира, будто соединены ethernet-кабелем.

Создание этой возможности кажется сложным; но для работающей упрощённой версии нужны всего два компонента, которые вместе дают результат, «доказуемый одним пингом»:

  • VSwitch — центральная зона принятия решений (учёт MAC, сопоставление с «портом», вещание). На Python.
  • VPort — порт соседнего устройства (интерфейс TAP + UDP-отправитель). На C.

Архитектура: ядро VSwitch работает на центральном хосте; каждый «конечный пользователь» запускает VPort, который через TAP соединён с ядром Linux и доставляет ethernet-кадры в VSwitch по UDP. Итог — менее 300 строк кода!

VPort — виртуальная сетевая карта на C (файл vport.c)

У VPort две главные задачи: (1) принимать L2-кадры из Linux через устройство TAP и (2) доставлять кадры в VSwitch по UDP (и наоборот). Со следующей структурой:

c#define TAP_DEV "/dev/net/tun"
#define UDP_PORT 9999

struct vport {
    int tap_fd;      /* fd устройства TAP (создаётся tun_alloc) */
    int udp_fd;      /* fd UDP-сокета, соединённого с VSwitch */
    struct sockaddr_in vswitch_addr;
    char mac[6];     /* фиктивный MAC этого порта */
c/* --- tun_alloc: создание интерфейса TAP -- */
int tun_alloc(char *dev) {
    struct ifreq ifr;
    int fd, err;

    if ((fd = open("/dev/net/tun", O_RDWR)) < 0)
        return -1;

    memset(&ifr, 0, sizeof(ifr));
    ifr.ifr_flags = IFF_TAP | IFF_NO_PI;   /* TAP и без packet-info */
    if (*dev) strncpy(ifr.ifr_name, dev, IFNAMSIZ);

    err = ioctl(fd, TUNSETIFF, (void *)&ifr);
    if (err < 0) { close(fd); return err; }
    strcpy(dev, ifr.ifr_name);
    return fd;
}

Главный цикл VPort: или read из TAP, или recvfrom из UDP; так как оба вызова блокирующие, используем select(), чтобы слушать оба одновременно:

cfor (;;) {
    fd_set rfds; FD_ZERO(&rfds);
    FD_SET(vp->tap_fd, &rfds); FD_SET(vp->udp_fd, &rfds);
    select(max_fd + 1, &rfds, NULL, NULL, NULL);

    if (FD_ISSET(vp->tap_fd, &rfds)) {
        /* Кадр пришёл из Linux: отправляем к VSwitch */
        n = read(vp->tap_fd, frame, sizeof frame);
        sendto(vp->udp_fd, frame, n, 0,
               (struct sockaddr *)&vp->vswitch_addr, sizeof vp->vswitch_addr);
    }

    if (FD_ISSET(vp->udp_fd, &rfds)) {
        /* Кадр пришёл от VSwitch: инжектируем в TAP */
        n = recvfrom(vp->udp_fd, frame, sizeof frame, 0, NULL, NULL);
        write(vp->tap_fd, frame, n);
    }
}

VSwitch — логика принятия решений на Python (файл vswitch.py)

Сам коммутатор тоже должен быть «портом»: для каждого VPort, который к нему подключается. Для простоты карта порт-MAC хранится в памяти:

python"""vswitch.py — беспечный L2-коммутатор."""
import socket

UDP_PORT = 9999
MAC_TABLE = {}          # mac → адрес (ip, port) порта
PORT_MAP  = {}          # (ip, port) → mac, виденные на этом порте

def main():
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    s.bind(("", UDP_PORT))
    print(f"VSwitch listening on UDP {UDP_PORT}")

    while True:
        frame, addr = s.recvfrom(65535)
        print(f"\nGot {len(frame)} bytes from {addr}")

        src_mac = frame[6:12].hex(":")
        dst_mac = frame[0:6].hex(":")

        # 1. Обучение: источник сопоставляется со своим портом
        MAC_TABLE[src_mac] = addr
        print(f"  Learn: {src_mac} → {addr}")

        if dst_mac == "ff:ff:ff:ff:ff:ff":
            # 2. Вещание: всем портам, кроме отправителя
            for a in set(MAC_TABLE.values()):
                if a != addr:
                    s.sendto(frame, a)
        elif dst_mac in MAC_TABLE:
            # 3. Одноадресная отправка: только порту назначения
            s.sendto(frame, MAC_TABLE[dst_mac])
        else:
            # 4. Неизвестно: как вещание, обучимся позже
            for a in set(MAC_TABLE.values()):
                if a != addr:
                    s.sendto(frame, a)

if __name__ == "__main__":
    main()

Развёртывание и тест: докажи пингом

Примечание: vport.c требует root для создания TAP; но когда TAP создан и ему выдан IP-адрес, функции read/write работают с обычными правами.

bash# 1. Запусти VSwitch на центральной машине
python3 vswitch.py

# 2. Собери и запусти VPort на каждой машине
make
sudo ./vport  # укажи адрес VSwitch в коде

# 3. Дай IP-адрес созданному интерфейсу TAP (например, l2vpn0)
sudo ip addr add 192.168.0.1/24 dev l2vpn0
sudo ip link set l2vpn0 up

# 4. Повтори то же на другой стороне и пингуй
ping 192.168.0.2

Если во время пинга вывод VSwitch показывает «Learn» (ответы ARP и ICMP), значит вся конструкция работает: две реальные машины общаются по UDP, как будто соединены ethernet-кабелем, — без какой-либо инфраструктуры посередине.

Что дальше

Оригинальный ZeroTier упаковывает ethernet-кадр в собственный туннельный протокол, а не в сырой UDP; и (главное) выполняет сквозное шифрование публичными ключами (curve25519) и шифром AES. В нашей версии любой может прочитать или изменить кадры! Идеи для эволюции: счётчик последовательности и auth-token против replay, таблица MAC с таймаутом и удалением старых записей, конкурентность через threads/async для масштабируемости и совместная поддержка IPv4/IPv6/Ethernet.

Пишем шелл на C

Оригинальный источник: «Write a Shell in C»

Введение — что такое шелл?

Шелл запускает процессы и объясняет им, где находятся их stdin и stdout. А для этого у него есть (скриптовый) язык. Значит, шелл делает две вещи: запускает программы через fork() и execvp() и создаёт язык — то есть «операторы» (statements) и «токены» (tokens).

В этом уроке мы построим минимальный шелл на C: чтение строки (read_line), разбор строки на аргументы (split_line), выполнение команды (execute — эту трудную часть делаем сразу, а потом переходим к структурам) и главный цикл, сшивающий всё вместе. Затем по порядку: встроенные команды, поиск пути (PATH), cd, фоновые процессы и Ctrl-C, и наконец полный запуск — и на каждом шаге проверяем через valgrind --leak-check=yes, что память не течёт.

Чтение строки команды

Сначала используем getline(), но позже при изучении токенизатора строим специальную структуру, которая хранит отдельно не только всю строку, но и каждый токен, чтобы в итоге стали возможны встроенные команды и фон:

c/* Структура командной строки и её частей */
typedef struct {
    char *buffer;
    size_t buf_len;
    int argc;
    char **argv;
    char *program_name; /* argv[0] без слешей, для сообщений об ошибках */
    int background;     /* должна ли эта команда выполняться в фоне? */
    int stdin, stdout;  /* перенаправление? */
} lsh_cmd;
c/* Простейший случай: читаем строку из stdin */
char *lsh_read_line(void) {
    char *line = NULL;
    ssize_t bufsize = 0;
    if (getline(&line, &bufsize, stdin) == -1) {
        if (feof(stdin)) exit(EXIT_SUCCESS); /* Ctrl-D */
        else { perror("readline"); exit(EXIT_FAILURE); }
    }
    return line;
}

Разбор строки на аргументы

Простой случай — разделитель-пробел и токенизация через strtok:

c#define LSH_TOK_BUFSIZE 64
#define LSH_TOK_DELIM " \t\r\n\a"

char **lsh_split_line(char *line) {
    int bufsize = LSH_TOK_BUFSIZE, position = 0;
    char **tokens = malloc(bufsize * sizeof(char*));
    char *token;

    if (!tokens) { fprintf(stderr, "lsh: allocation error\n"); exit(EXIT_FAILURE); }

    token = strtok(line, LSH_TOK_DELIM);
    while (token != NULL) {
        tokens[position] = token;
        position++;
        if (position >= bufsize) { /* рост буфера */ }
        token = strtok(NULL, LSH_TOK_DELIM);
    }
    tokens[position] = NULL;
    return tokens;
}

А в финальной версии — токенизатор с выделением специальных символов >, >>, <, |, & (фон) и построением объекта lsh_cmd, который охватывает и перенаправление, и фон.

Выполнение — сердце шелла

Дочерний процесс создаётся через fork(); дочерний применяет перенаправление и запускает программу через execvp; если шелл в обычном режиме, waitpid ожидает (это и есть «приглашение возвращается»), а если фон — шелл немедленно печатает следующее приглашение:

cint lsh_launch(char **args) {
    pid_t pid, wpid;
    int status;

    pid = fork();
    if (pid == 0) {
        /* Дочерний: exec не возвращается, кроме случая ошибки */
        if (execvp(args[0], args) == -1) perror("lsh");
        exit(EXIT_FAILURE);
    } else if (pid < 0) {
        perror("lsh");
    } else {
        do {
            wpid = waitpid(pid, &status, WUNTRACED);
        } while (!WIFEXITED(status) && !WIFSIGNALED(status));
    }
    return 1;
}

Встроенные команды и поиск программ

Задачи шелла (вроде cd, exit, help) должны выполняться без создания процесса; поэтому храним список имён с соответствующими функциями (lsh_builtin_str[] и lsh_builtin_func[]). Для поиска программ ходим по путям PATH; сначала проверяем, содержит ли имя /; если нет, через strchr строим для каждого пути PATH вариант /name и проверяем существование через stat. execvp делает это за нас, но чтение кода поиска PATH помогает понять глубже: отделите пути через : из getenv("PATH"), соберите их и инкапсулируйте через strdup.

Реализация cd — занятная конструкция: chdir() действует только в текущем процессе, поэтому это должна быть встроенная команда — обычной программой она быть не может, ведь chdir меняет каталог дочернего процесса, а не самого шелла!

cint lsh_cd(char **args) {
    if (args[1] == NULL) {
        fprintf(stderr, "lsh: expected argument to \"cd\"\n");
    } else if (chdir(args[1]) != 0) {
        perror("lsh");
    }
    return 1;
}

Фоновые процессы и Ctrl-C

Фон означает: если команда заканчивается на & (lsh->background), для дочернего вызываем setpgid(0,0), чтобы он стал независимым от терминала (нажатие Ctrl-C не убивает шелл, ведь сигнал уходит только процессам той же группы), и waitpid(pid, &status, WNOHANG) определяет, жив ли ещё процесс. Утечки ищет valgrind --leak-check=yes ./lsh, запускаемый на каждом шаге; убеждаемся, что все malloc освобождаются (free_list для структур команд).

Полный код и основной цикл

cint main(int argc, char **argv) {
    /* загрузка конфигурации и выполнение команд запуска */
    lsh_loop();
    return EXIT_SUCCESS;
}

void lsh_loop(void) {
    char *line; char **args; int status;
    do {
        printf("> ");
        line = lsh_read_line();
        args = lsh_split_line(line);
        status = lsh_execute(args);
        free(line); free(args);
    } while (status);
}

(В полной версии lsh_execute распознаёт встроенные команды и вызывает lsh_launch; цикл работает со структурой lsh_cmd.) Весь код — около 500 строк чистого C — доступен на github.com/brenns10/lsh. Для дальнейшего изучения продолжением этой статьи служит её вторая часть.
